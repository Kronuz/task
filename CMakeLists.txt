cmake_minimum_required(VERSION 3.14)
project(task LANGUAGES CXX)

include(FetchContent)
find_package(Threads REQUIRED)

# Header-only: task.hh needs only the standard library and threads (sync_wait blocks).
add_library(task INTERFACE)
target_link_libraries(task INTERFACE Threads::Threads)
target_include_directories(task INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(task INTERFACE cxx_std_20)
add_library(task::task ALIAS task)

# task_asio.hh binds tasks to standalone Asio (the same version Kronuz/reactor uses).
option(TASK_ASIO "Build the task::asio target" ON)
if(TASK_ASIO)
  set(ASIO_INCLUDE_DIR "" CACHE PATH "use an existing standalone Asio include dir instead of fetching")
  if(ASIO_INCLUDE_DIR STREQUAL "")
    FetchContent_Declare(asio GIT_REPOSITORY https://github.com/chriskohlhoff/asio.git GIT_TAG asio-1-36-0)
    FetchContent_MakeAvailable(asio)
    set(ASIO_INCLUDE_DIR "${asio_SOURCE_DIR}/asio/include")
  endif()
  add_library(task_asio INTERFACE)
  target_include_directories(task_asio INTERFACE "${ASIO_INCLUDE_DIR}")
  target_compile_definitions(task_asio INTERFACE ASIO_STANDALONE)
  target_link_libraries(task_asio INTERFACE task Threads::Threads)
  add_library(task::asio ALIAS task_asio)
endif()

# Tests and benchmarks build only when this repo is the top-level project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  add_executable(task_test test/test.cc)
  target_link_libraries(task_test PRIVATE task Threads::Threads)
  add_test(NAME task COMMAND task_test)
  add_executable(task_sync_test test/sync_test.cc)
  target_link_libraries(task_sync_test PRIVATE task Threads::Threads)
  add_test(NAME task_sync COMMAND task_sync_test)
  add_executable(task_bench bench/bench.cc)
  target_link_libraries(task_bench PRIVATE task)
  if(TASK_ASIO)
    add_executable(task_asio_test test/asio_test.cc)
    target_link_libraries(task_asio_test PRIVATE task::asio)
    add_test(NAME task_asio COMMAND task_asio_test)
    target_link_libraries(task_bench PRIVATE task::asio)
    target_compile_definitions(task_bench PRIVATE TASK_BENCH_ASIO=1)
  endif()
endif()
